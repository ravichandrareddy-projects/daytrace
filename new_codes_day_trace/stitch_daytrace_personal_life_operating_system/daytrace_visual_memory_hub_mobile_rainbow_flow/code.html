LeetCode 297: Serialize Binary Tree

class Codec:
    def serialize(self, root):
        vals = []
        def doit(node):
            if node:
                vals.append(str(node.val))
                doit(node.left); doit(node.right)
            else: vals.append("#")
        doit(root)
        return " ".join(vals)
